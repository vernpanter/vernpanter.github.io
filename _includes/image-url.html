{%- comment -%} Sets "thumb" and "large" for include.path. The resized copies are used when they were made, otherwise both are the uploaded file. {%- endcomment -%} {%- assign image = site.data.images[include.path] -%} {%- if image -%} {%- assign thumb = image.thumb -%} {%- assign large = image.large -%} {%- else -%} {%- assign thumb = include.path | uri_escape | replace: "#", "%23" | replace: "?", "%3F" -%} {%- assign large = thumb -%} {%- endif -%}
